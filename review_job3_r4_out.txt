   156	    /// Docs/architecture.md §8: on daemon start, mark every **non-terminal**
   157	    /// instance (`queued` / `loading` / `ready` / `draining` / `unloading`)
   158	    /// as `crashed` — a process was expected to be running but the daemon
   159	    /// restarted, so those records are orphaned. Each move is validated
   160	    /// through the domain state machine (every non-terminal state legally
   161	    /// reaches `crashed`) and audit-logged, and the whole recovery commits
   162	    /// in one transaction. Settled records (`unloaded` / `failed` /
   163	    /// `crashed`) are left untouched, and a second call finds nothing to do.
   164	    ///
   165	    /// Returns the instances as they were found (with their pre-recovery
   166	    /// state), in recovery order. Non-terminal **operations** are exposed
   167	    /// via `crate::repos::OperationsRepo::non_terminal` for the supervisor
   168	    /// (job 6) rather than settled here.
   169	    ///
   170	    /// # Errors
   171	    ///
   172	    /// `InvalidStateTransition` if a stored state cannot legally reach
   173	    /// `crashed` (defensive: schema fence and state machine agree on the
   174	    /// vocabulary), or `Internal` on storage failure.
   175	    pub async fn recover_non_terminal_instances(&self) -> Result<Vec<Instance>> {
   176	        let stale = InstancesRepo::non_terminal(self.pool()).await?;
   177	        if stale.is_empty() {
   178	            return Ok(Vec::new());
   179	        }
   180	        let at = now();
   181	        let mut tx = self.transaction().await?;
   182	        for found in &stale {
   183	            // Re-read the row inside the transaction: the pool-based scan
   184	            // above predates it, and a legal `ready -> draining -> ready`
   185	            // cycle could have moved the row and back. Validate the move
   186	            // through the frozen domain state machine (never a raw write)
   187	            // against this fresh state, then guard the update on BOTH the
   188	            // state and the `updated_at` revision so a same-state ABA
   189	            // cycle is rejected too (`&mut SqliteConnection` is not `Copy`,
   190	            // so the two-query repository guard cannot run in a transaction;
   191	            // sequential re-borrows of the transaction do the same job).

   193	                sqlx::query("SELECT state, updated_at FROM instances WHERE instance_id = ?1")
   194	                    .bind(found.instance_id())

   197	                    .map_err(|e| storage_error(&e, "read instance state"))?;

   199	                return Err(DomainError::with_message(
   200	                    ErrorCode::Internal,
   201	                    format!(
   202	                        "instance {} vanished before recovery applied",
   203	                        found.instance_id()

   205	                ));
   206	            };

   208	                DomainError::with_message(
   209	                    ErrorCode::Internal,
   210	                    format!("read instance {} state: {e}", found.instance_id()),
   211	                )
   212	            })?;
   213	            let revised = current.try_get::<String, _>(1).map_err(|e| {
   214	                DomainError::with_message(
   215	                    ErrorCode::Internal,
   216	                    format!("read instance {} updated_at: {e}", found.instance_id()),
   217	                )
   218	            })?;
   219	            let current_state = parse_wire::<InstanceState>(&raw_state, "instances.state")?;
   220	            transition_instance(current_state, InstanceState::Crashed).map_err(|_| {
   221	                DomainError::with_message(
   222	                    ErrorCode::InvalidStateTransition,
   223	                    format!(
   224	                        "instance {} in {current_state:?} cannot legally reach crashed",
   225	                        found.instance_id()
   226	                    ),
   227	                )
   228	            })?;
   229	            let result = sqlx::query(
   230	                "UPDATE instances SET state = ?2, failure = NULL, updated_at = ?3 \
   231	                 WHERE instance_id = ?1 AND state = ?4 AND updated_at = ?5",
   232	            )
   233	            .bind(found.instance_id())
   234	            .bind(wire_token(&InstanceState::Crashed))
   235	            .bind(ts_string(at))
   236	            .bind(wire_token(&current_state))
   237	            .bind(&revised)
   238	            .execute(&mut *tx)
   239	            .await
   240	            .map_err(|e| storage_error(&e, "write instance state"))?;
   241	            if result.rows_affected() == 0 {
   242	                return Err(DomainError::with_message(
   243	                    ErrorCode::InvalidStateTransition,
   244	                    format!(
   245	                        "instance {} changed state before recovery applied",
   246	                        found.instance_id()
   247	                    ),
   248	                ));
   249	            }
   250	            AuditRepo::append(
   251	                &mut *tx,
   252	                &AuditEvent::new(
   253	                    AuditKind::InstanceRecovered,
   254	                    Some("instance".to_owned()),
   255	                    Some(found.instance_id().to_string()),
   256	                    None,
   257	                    serde_json::json!({ "previous_state": found.state() }),
   258	                    at,
   259	                ),
   260	            )
   120	        ],
   121	        _ => panic!("test needs a non-terminal state"),
   122	    };
   123	    for step in path {
   124	        instance = instance.with_state(*step).expect("test path is legal");
   125	    }
   126	    instance
