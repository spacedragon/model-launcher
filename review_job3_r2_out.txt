    19	/// Ensure the on-disk database file is owner-only (`0600`), per
    20	/// `docs/security-and-deployment.md` (the file holds credentials-adjacent
    21	/// state and must not be group/world readable).
    22	///
    23	/// If the file does not yet exist it is created owner-only up front (so the
    24	/// first byte SQLite writes already lands in a `0600` file — no permissive
    25	/// window); if it exists and is more permissive than `0600` (any group/other
    26	/// bit set) it is tightened to `0600`. The `set_permissions` call is what makes
    27	/// the result deterministic regardless of the process `umask`.
    28	///
    29	/// Unix-only: on Windows the file-attribute model differs and this is a no-op.
    30	///
    31	/// # Errors
    32	///
    33	/// [`ErrorCode::Internal`] if the file cannot be stat-ed, created, or `chmod`-ed.
    34	#[cfg(unix)]
    35	fn secure_db_file(path: &Path) -> Result<()> {
    36	    use std::os::unix::fs::{OpenOptionsExt, PermissionsExt};
    37	    let meta = match std::fs::metadata(path) {
    38	        Ok(meta) => meta,
    39	        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
    40	            std::fs::OpenOptions::new()
    41	                .write(true)
    42	                .create(true)
    43	                .mode(0o600)
    44	                .open(path)
    45	                .map_err(|e| {
    46	                    DomainError::with_message(
    47	                        ErrorCode::Internal,
    48	                        format!("create database file {path:?}: {e}"),
    49	                    )
    50	                })?;
    51	            std::fs::metadata(path).map_err(|e| {
    52	                DomainError::with_message(
    53	                    ErrorCode::Internal,
    54	                    format!("stat database file {path:?}: {e}"),
    55	                )
    56	            })?
    57	        }
    58	        Err(e) => {
    59	            return Err(DomainError::with_message(
    60	                ErrorCode::Internal,
    61	                format!("stat database file {path:?}: {e}"),
    62	            ));
    63	        }
    64	    };
    65	    // Deterministically force owner-only (0600) regardless of umask.
    66	    if meta.permissions().mode() & 0o777 != 0o600 {
    67	        std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600)).map_err(|e| {
    68	            DomainError::with_message(
    69	                ErrorCode::Internal,
    70	                format!("set database file {path:?} permissions to 0600: {e}"),
    71	            )
    72	        })?;
    73	    }
    74	    Ok(())
    75	}

   155	    /// Docs/architecture.md §8: on daemon start, mark every **non-terminal**
   156	    /// instance (`queued` / `loading` / `ready` / `draining` / `unloading`)
   157	    /// as `crashed` — a process was expected to be running but the daemon
   158	    /// restarted, so those records are orphaned. Each move is validated
   159	    /// through the domain state machine (every non-terminal state legally
   160	    /// reaches `crashed`) and audit-logged, and the whole recovery commits
   161	    /// in one transaction. Settled records (`unloaded` / `failed` /
   162	    /// `crashed`) are left untouched, and a second call finds nothing to do.
   163	    ///
   164	    /// Returns the instances as they were found (with their pre-recovery
   165	    /// state), in recovery order. Non-terminal **operations** are exposed
   166	    /// via `crate::repos::OperationsRepo::non_terminal` for the supervisor
   167	    /// (job 6) rather than settled here.
   168	    ///
   169	    /// # Errors
   170	    ///
   171	    /// `InvalidStateTransition` if a stored state cannot legally reach
   172	    /// `crashed` (defensive: schema fence and state machine agree on the
   173	    /// vocabulary), or `Internal` on storage failure.
   174	    pub async fn recover_non_terminal_instances(&self) -> Result<Vec<Instance>> {
   175	        let stale = InstancesRepo::non_terminal(self.pool()).await?;
   176	        if stale.is_empty() {
   177	            return Ok(Vec::new());
   178	        }
   179	        let at = now();
   180	        let mut tx = self.transaction().await?;
   181	        for found in &stale {
   182	            // Validate the move through the frozen domain state machine (never
   183	            // a raw write). The expected `from` is the state just read, so the
   184	            // guarded update below needs no re-SELECT on the transaction
   185	            // connection (`&mut SqliteConnection` is not `Copy`, so the

   193	                        found.state()
   194	                    ),
   195	                )
   196	            })?;
   197	            let result = sqlx::query(
   198	                "UPDATE instances SET state = ?2, failure = NULL, updated_at = ?3 \
   199	                 WHERE instance_id = ?1 AND state = ?4",
   200	            )
   201	            .bind(found.instance_id())
   202	            .bind(wire_token(&InstanceState::Crashed))
   203	            .bind(ts_string(at))
   204	            .bind(wire_token(&found.state()))
   205	            .execute(&mut *tx)
   206	            .await
   207	            .map_err(|e| storage_error(&e, "write instance state"))?;
   208	            if result.rows_affected() == 0 {
   209	                return Err(DomainError::with_message(
   210	                    ErrorCode::InvalidStateTransition,
   211	                    format!(
   212	                        "instance {} changed state before recovery applied",
   213	                        found.instance_id()
   214	                    ),
   215	                ));
   216	            }
   217	            AuditRepo::append(
   218	                &mut *tx,
   219	                &AuditEvent::new(
   220	                    AuditKind::InstanceRecovered,
   221	                    Some("instance".to_owned()),
   222	                    Some(found.instance_id().to_string()),
   223	                    None,
   224	                    serde_json::json!({ "previous_state": found.state() }),
   225	                    at,
   226	                ),
   227	            )
   228	            .await?;
   229	        }
   230	        tx.commit()
   231	            .await
   232	            .map_err(|e| storage_error(&e, "commit recovery transaction"))?;
   233	        Ok(stale)
   234	    }

   136	/// Translate a `sqlx::Error` into the domain error catalog (see module docs).
   137	#[must_use]
   138	pub(crate) fn storage_error(err: &sqlx::Error, context: &str) -> DomainError {
   139	    match err {
   140	        sqlx::Error::RowNotFound => {
   141	            DomainError::with_message(ErrorCode::Internal, format!("{context}: row not found"))
   142	        }
   143	        sqlx::Error::Database(db) => {
   144	            // `sqlx::DatabaseError` exposes no structured constraint kind for
   145	            // SQLite in 0.8, so the message is the contract: every SQLite
   146	            // constraint violation is reported as "... constraint failed: ...".
   147	            let message = db.message();
   148	            if message.contains("CHECK constraint failed") {
   149	                // Only the *named* state fences are state-machine violations;
   150	                // every other CHECK (unnamed kind fences, value ranges,
   151	                // booleans) is a data-integrity problem -> `InvalidRequest`.
   152	                let fence = STATE_FENCE_CONSTRAINTS
   153	                    .iter()
   154	                    .find(|name| message.contains(*name));
   155	                match fence {
   156	                    Some(name) => DomainError::with_message(
   157	                        ErrorCode::InvalidStateTransition,
   158	                        format!("{context}: state rejected by the {name} constraint"),
   159	                    ),
   160	                    None => DomainError::with_message(
   161	                        ErrorCode::InvalidRequest,
   162	                        format!("{context}: check constraint violated: {message}"),
   163	                    ),

     1	//! Pure state machines: the instance load-state machine and the operation
     2	//! state machine (`docs/architecture.md` §3, §5).
     3	//!
     4	//! Both are expressed as explicit target-state transition tables so every legal
     5	//! move is visible in one place and illegal moves are rejected deterministically.
     6	//! `can_transition` / `transition` are pure functions — no I/O, no mutation of
     7	//! the caller's state — so they are trivially unit-testable and can be reused
     8	//! by the runtime supervisor, the scheduler, and the API layer.
     9	//!
    10	//! ## Instance state machine (`docs/architecture.md` §5)
    11	//!
    12	//! ```text
    13	//! unloaded -> queued -> loading -> ready
    14	//!                    |          |
    15	//!                    v          v
    16	//!                  failed    draining -> unloading -> unloaded
    17	//!                                   |
    18	//!                                   v
    19	//!                                 ready   (drain cancelled / timed-out fallback)
    20	//!
    21	//! any running state --unexpected exit--> crashed
    22	//! crashed --explicit load----------------> queued
    23	//! ```
    24	//!
    25	//! Notes on the encoding choices (documented, and asserted in tests):
    26	//! - Only `crashed` re-queues on an explicit load
    27	//!   (`crashed --explicit load--> queued`). `failed` is a settled terminal
    28	//!   with **no** re-queue edge: a later load starts a fresh instance
    29	//!   (`docs/architecture.md` §5). `loading` is never a dead-end — it resolves
    30	//!   to `ready`/`failed`/`crashed`/`unloading` — which is what
    31	//!   `docs/development-plan.md` §5 "不出现永久 `loading`" means.
    32	//! - A load that never reaches `ready` ends in `failed`; an unexpected exit of
    33	//!   a serving (`ready`) process ends in `crashed`. A spawned-but-not-ready
    34	//!   process may also be torn down via `unloading`.
    35	//! - `draining -> ready` is the drain-cancelled / timed-out fallback.
    36	//! - Any running state (`loading`/`ready`/`draining`) may reach `crashed` on an
    37	//!   unexpected child exit (`docs/architecture.md` §5: 任意运行态 → crashed), so
    38	//!   `draining` (still a live process serving existing requests) also crashes.
    39	//! - On daemon restart, recovery marks *every non-terminal* state
    40	//!   (`queued`/`loading`/`ready`/`draining`/`unloading`) `crashed` — a process
    41	//!   was expected to be running but is now orphaned (`docs/architecture.md`
    42	//!   §8). The table therefore lets every non-terminal state reach `crashed` so
    43	//!   recovery uses the validated path, not a raw state write.
    44

    48	impl InstanceState {
    49	    /// Target states reachable from `self` in a single step.
    50	    const fn transitions(self) -> &'static [InstanceState] {
    51	        use InstanceState as S;
    52	        match self {
    53	            S::Queued => &[S::Loading, S::Unloaded, S::Failed, S::Crashed],
    54	            S::Loading => &[S::Ready, S::Failed, S::Unloading, S::Crashed],
    55	            S::Ready => &[S::Draining, S::Crashed],
    56	            S::Draining => &[S::Unloading, S::Ready, S::Crashed],
    57	            S::Unloading => &[S::Unloaded, S::Crashed],
    58	            S::Unloaded | S::Crashed => &[S::Queued],
    59	            S::Failed => &[],
    60	        }
    61	    }
    62	
    63	    /// Whether moving from `self` to `target` is a legal single step.
    64	    #[must_use]
    65	    pub fn can_transition_to(self, target: InstanceState) -> bool {
    66	        self.transitions().contains(&target)
    67	    }
    68	
    69	    /// Whether the instance lifecycle is settled (no in-flight process).
    70	    ///
    71	    /// This is the property `docs/architecture.md` §8 needs: on daemon start,
    72	    /// records whose state is *not* terminal — a process was expected to be
    73	    /// running (`queued`/`loading`/`ready`/`draining`/`unloading`) — are
    74	    /// marked `crashed`/recovery-required, while `unloaded`/`failed`/`crashed`
    75	    /// are already settled and left alone. It is deliberately NOT "no outgoing
    76	    /// transition": `unloaded` can be re-queued, so every state has an
    77	    /// outgoing edge and edge-based terminality is always false, which would
    78	    /// wrongly flag clean/historical records for recovery.
    79	    #[must_use]
    80	    pub const fn is_terminal(self) -> bool {
    81	        matches!(self, Self::Unloaded | Self::Failed | Self::Crashed)
    82	    }

    84	    /// The states that can legally follow a `ready` instance (drain / crash).
    85	    #[must_use]
    86	    pub const fn ready_followers() -> &'static [InstanceState] {
    87	        InstanceState::Ready.transitions()
    88	    }

   116	///
   117	/// Returns the (unchanged) target on success, or a
   118	/// [`ErrorCode::InvalidStateTransition`] error on an illegal move. Pure: it
   119	/// does not mutate anything; the caller applies the returned state.

   123	/// [`ErrorCode::InvalidStateTransition`] when `to` is not reachable from

   131	            format!("instance cannot move from {from:?} to {to:?}"),
   132	        ))
   133	    }

   142	///
   143	/// [`ErrorCode::InvalidStateTransition`] when `to` is not reachable from
   144	/// `from` in the operation transition table.

   148	    } else {

   151	            format!("operation cannot move from {from:?} to {to:?}"),

   153	    }

    60	/// exists, runs `transition` (the domain machine) on `(current, to)`. A
    61	/// transition the domain machine rejects returns `DomainError` with
    62	/// `ErrorCode::InvalidStateTransition`; an undecodable stored state is surfaced
    63	/// as `Internal`.
    64	///
    65	/// Returns the **current** state when the row exists, or `None` for a
    66	/// first-seen row (no prior state, so a legal initial state may be written and
    67	/// no transition check applies). Callers then issue a write guarded on the
    68	/// returned state (`UPDATE ... WHERE <state_col> = <returned>`). Because the
    69	/// SELECT and the write must run on the same executor, callers pass a
    70	/// re-usable executor (a pool handle, `Copy`); atomic multi-table guarded
    71	/// writes are performed inline on a transaction connection.
    72	async fn validate_state<'c, E, S>(
    73	    exec: E,
    74	    table: &str,
    75	    key_col: &str,
    76	    key: &str,
    77	    to: S,
    78	    transition: fn(S, S) -> Result<S>,
    79	) -> Result<Option<S>>
    80	where
    81	    E: sqlx::Executor<'c, Database = Sqlite> + 'c,
    82	    S: std::fmt::Debug + Copy + Send + serde::de::DeserializeOwned + 'c,
    83	{
    84	    let sql = format!("SELECT state FROM {table} WHERE {key_col} = ?1");
    85	    let current: Option<String> = sqlx::query_scalar(&sql)
    86	        .bind(key)
    87	        .fetch_optional(exec)
    88	        .await
    89	        .map_err(|e| storage_error(&e, &format!("read {table} state")))?;
    90	    match current {
    91	        None => Ok(None),
    92	        Some(raw) => {
    93	            let current = parse_wire::<S>(&raw, &format!("{table}.state"))?;
    94	            transition(current, to)?;
    95	            Ok(Some(current))
    96	        }
    97	    }
    98	}

  1151	/// Rebuild a domain `Instance` from a stored row.
  1152	///
  1153	/// The stored state is reached from the initial `unloaded` state by a
  1154	/// validated path in the domain state machine (every state in the vocabulary
  1155	/// has one — see `state_machine.rs`), so the reconstruction never bypasses
  1156	/// the state machine.
  1157	fn instance_from_row(row: InstanceRow) -> Result<Instance> {
  1158	    let load_config = from_json::<LoadConfig>(&row.load_config, "load_config")?;
  1159	    let state = parse_wire::<InstanceState>(&row.state, "state")?;
  1160	    let mut instance = Instance::new(row.instance_id, row.model_id, row.runtime_id, load_config);
  1161	    for step in reach_path(state) {
  1162	        instance = instance.with_state(*step)?;
  1163	    }
  1164	    // `pid` / `port` are only trusted once the supervisor confirms them, so a
  1165	    // SQL NULL stays `None`; but a non-NULL value that is out of the column's
  1166	    // unsigned domain is data corruption and must surface `Internal`, never be
  1167	    // silently dropped to `None` (`docs/architecture.md` §3).
  1168	    instance.pid = match row.pid {
  1169	        None => None,
  1170	        Some(v) => Some(from_i64::<u32>(v, "pid")?),
  1171	    };
  1172	    instance.port = match row.port {
  1173	        None => None,
  1174	        Some(v) => Some(from_i64::<u16>(v, "port")?),
  1175	    };
  1176	    instance.device_ids = from_json::<Vec<u32>>(&row.device_ids, "device_ids")?;
  1177	    if let Some(raw) = &row.started_at {
  1178	        instance.started_at = Some(parse_ts(raw, "started_at")?);
  1179	    }
  1180	    if let Some(raw) = &row.last_used_at {
  1181	        instance.last_used_at = Some(parse_ts(raw, "last_used_at")?);
  1182	    }
  1183	    instance.active_requests = from_i64(row.active_requests, "active_requests")?;
  1184	    instance.health = match row.health {
  1185	        Some(raw) => Some(from_json::<InstanceHealth>(&raw, "health")?),
  1186	        None => None,
  1187	    };
  1188	    instance.failure = match row.failure {
  1189	        Some(raw) => Some(from_json::<InstanceFailure>(&raw, "failure")?),
  1190	        None => None,
  1191	    };
  1192	    Ok(instance)
  1193	}

  1205	        S::Unloading => &[S::Queued, S::Loading, S::Unloading],
  1206	        S::Failed => &[S::Queued, S::Failed],
  1207	        S::Crashed => &[S::Queued, S::Crashed],

  1405	    /// Every operation in a non-terminal state (`queued` / `running`), for
  1406	    /// the supervisor to settle on daemon restart (M1 job 6). Tokens are
  1407	    /// derived from `OperationState::is_terminal` so the two cannot drift.
  1408	    ///
  1409	    /// # Errors
  1410	    ///
  1411	    /// `Internal` on a corrupt row or storage failure.
  1412	    pub async fn non_terminal<'c, E>(exec: E) -> Result<Vec<Operation>>
  1413	    where
  1414	        E: sqlx::Executor<'c, Database = Sqlite> + 'c,
  1415	    {
  1416	        let all = [
  1417	            OperationState::Queued,
  1418	            OperationState::Running,
  1419	            OperationState::Succeeded,
  1420	            OperationState::Failed,
  1421	            OperationState::Cancelled,
  1422	        ];
  1423	        let states = all
  1424	            .into_iter()
  1425	            .filter(|s| !s.is_terminal())
  1426	            .map(|s| format!("{:?}", wire_token(&s)))
  1427	            .collect::<Vec<_>>()
  1428	            .join(", ");
  1429	        let sql = format!(
  1430	            "SELECT operation_id, kind, state, instance_id, model_id, \
  1431	             created_at, finished_at, error, result FROM operations \
  1432	             WHERE state IN ({states}) ORDER BY created_at, operation_id"
  1433	        );
  1434	        let rows: Vec<OperationRow> = sqlx::query_as(&sql)
  1435	            .fetch_all(exec)
  1436	            .await
  1437	            .map_err(|e| storage_error(&e, "query non-terminal operations"))?;
  1438	        rows.into_iter().map(operation_from_row).collect()
  1439	    }
  1440	}

  1444	    operation_id: String,
  1445	    kind: String,
  1446	    state: String,
  1447	    instance_id: Option<String>,
  1448	    model_id: Option<String>,
  1449	    created_at: String,
  1450	    finished_at: Option<String>,
  1451	    error: Option<String>,
  1452	    result: Option<String>,
  1453	}

  1455	/// Rebuild a domain `Operation` from a stored row (the state is reached from
  1456	/// the initial `queued` state by the validated path; the other fields are
  1457	/// public and assigned directly).
  1458	fn operation_from_row(row: OperationRow) -> Result<Operation> {
  1459	    let kind = parse_wire::<OperationKind>(&row.kind, "kind")?;
  1460	    let state = parse_wire::<OperationState>(&row.state, "state")?;
  1461	    let mut operation = Operation::new(row.operation_id, kind);
  1462	    for step in operation_reach_path(state) {
  1463	        operation = operation.with_state(*step)?;
  1464	    }
  1465	    operation.instance_id = row.instance_id;
  1466	    operation.model_id = row.model_id;
  1467	    operation.created_at = Some(parse_ts(&row.created_at, "created_at")?);
  1468	    operation.finished_at = match row.finished_at {
  1469	        Some(raw) => Some(parse_ts(&raw, "finished_at")?),
  1470	        None => None,
  1471	    };
  1472	    operation.error = match row.error {
  1473	        Some(raw) => Some(from_json::<OperationError>(&raw, "error")?),
  1474	        None => None,
  1475	    };
  1476	    operation.result = match row.result {
  1477	        Some(raw) => Some(from_json::<serde_json::Value>(&raw, "result")?),
  1478	        None => None,
  1479	    };
  1480	    Ok(operation)
  1481	}

  1483	/// The validated single-step path from the initial `queued` state to
  1484	/// `target` (each entry is legal from the previous; `state_machine.rs`).
  1485	const fn operation_reach_path(target: OperationState) -> &'static [OperationState] {
  1486	    use OperationState as S;
  1487	    match target {
  1488	        S::Queued => &[],
  1489	        S::Running => &[S::Running],
  1490	        S::Succeeded => &[S::Running, S::Succeeded],
  1491	        S::Failed => &[S::Running, S::Failed],
  1492	        S::Cancelled => &[S::Cancelled],
  1493	    }
  1494	}

   154	/// The `store` half of the §9 atomic write inside an open transaction: the
   155	/// trigger operation, the instance desired-state drive, and the audit event.
   156	async fn drive_desired_state_in_tx(
   157	    tx: &mut sqlx::SqliteTransaction<'static>,
   158	    instance: &Instance,
   159	    operation: &Operation,
   160	    desired: InstanceState,
   161	) {
   162	    // sqlx 0.8 implements `Executor` for `&mut SqliteConnection`, not for
   163	    // `&mut Transaction` (sqlx-core#3857): deref the transaction down to its
   164	    // connection before handing it to the repositories.
   165	    let connection = &mut **tx;
   166	    let at = Utc::now();
   167	    OperationsRepo::create(&mut *connection, operation)
   168	        .await
   169	        .expect("operation insert in tx");
   170	    InstancesRepo::set_desired_state(&mut *connection, &instance.instance_id, desired)
   171	        .await
   172	        .expect("desired state update in tx");
   173	    AuditRepo::append(
   174	        &mut *connection,
   175	        &AuditEvent::new(
   176	            AuditKind::InstanceDesiredChanged,
   177	            Some("instance".into()),
   178	            Some(instance.instance_id.clone()),
   179	            Some(operation.operation_id.clone()),
   180	            serde_json::json!({ "state": desired }),
   181	            at,
   182	        ),
   183	    )
   184	    .await
   185	    .expect("audit append in tx");
   186	}

-- 0001_initial.sql
--
-- Initial schema for the `model-serving` control plane (docs/architecture.md
-- 搂3, 搂8, 搂9).
--
-- Migration policy: **append-only**. Never edit or delete a shipped file; the
-- upgrade path for existing databases is "run the new numbered files".
--
-- Conventions (docs/api.md 搂1, 搂2.2):
--   * every state / kind / class column stores the lowercase snake_case wire
--     token as TEXT and is fenced by a named CHECK constraint so a row outside
--     the state-machine vocabulary can never be stored (docs/architecture.md
--     搂9; the repositories additionally validate transitions through the
--     domain state machines before writing);
--   * timestamps are RFC 3339 UTC TEXT (`created_at`, `updated_at`, `mtime`,
--     `started_at`, ...);
--   * structured payloads (`load_config`, `capabilities`, `metadata`,
--     `health`, `failure`, `device_ids`, `error`, `result`, `payload`) are
--     JSON TEXT.
--
-- NOTE: `PRAGMA journal_mode=WAL` and `PRAGMA foreign_keys=ON` are enforced at
-- connection time by `SqliteStore::open` (journal_mode cannot be switched
-- inside a transaction, so it must not live in a migration file).

-- Daemon-wide key/value settings (docs/api.md 搂5: GET/PATCH /admin/v1/settings).
-- `restart_required` flags values that only take effect after a restart.
CREATE TABLE settings (
    key              TEXT PRIMARY KEY,
    value            TEXT NOT NULL,
    restart_required INTEGER NOT NULL DEFAULT 0 CHECK (restart_required IN (0, 1)),
    updated_at       TEXT NOT NULL
);

-- Registered inference runtimes (docs/architecture.md 搂3 "Runtime").
-- `last_probe_ok` / `last_probed_at` hold the most recent probe outcome; they
-- are written by the runtime-probe job (M1 job 5) and preserved by CRUD.
CREATE TABLE runtimes (
    id             TEXT PRIMARY KEY,
    kind           TEXT NOT NULL CHECK (kind IN ('llama_cpp', 'ninfer')),
    executable_path TEXT NOT NULL,
    enabled        INTEGER NOT NULL DEFAULT 0 CHECK (enabled IN (0, 1)),
    version_text   TEXT,
    capabilities   TEXT NOT NULL DEFAULT '{}',
    fixed_args     TEXT NOT NULL DEFAULT '[]',
    last_probe_ok  INTEGER,
    last_probed_at TEXT
);

-- Controlled scan directories (docs/api.md 搂5: model-roots).
-- `last_scan_at` / `last_scan_result` hold the most recent scan outcome for
-- `GET /admin/v1/model-roots` (written by the scanner job, M1 job 4).
CREATE TABLE model_roots (
    id               TEXT PRIMARY KEY,
    path             TEXT NOT NULL UNIQUE,
    enabled          INTEGER NOT NULL DEFAULT 1 CHECK (enabled IN (0, 1)),
    last_scan_at     TEXT,
    last_scan_result TEXT,
    created_at       TEXT NOT NULL
);

-- Indexed model artifacts (docs/architecture.md 搂3 "Model").
-- `root_id` / `first_seen_at` / `last_seen_at` are scanner bookkeeping
-- reserved for the scan job (M1 job 4); CRUD upserts preserve them.
CREATE TABLE models (
    id                  TEXT PRIMARY KEY,
    key                 TEXT NOT NULL UNIQUE,
    path                TEXT NOT NULL,
    display_name        TEXT,
    artifact_kind       TEXT NOT NULL CHECK (artifact_kind IN ('gguf', 'ninfer')),
    size_bytes          INTEGER NOT NULL CHECK (size_bytes >= 0),
    mtime               TEXT NOT NULL,
    default_runtime_id  TEXT REFERENCES runtimes (id) ON DELETE SET NULL,
    default_load_config TEXT,
    metadata            TEXT,
    deleted             INTEGER NOT NULL DEFAULT 0 CHECK (deleted IN (0, 1)),
    root_id             TEXT REFERENCES model_roots (id) ON DELETE SET NULL,
    first_seen_at       TEXT NOT NULL,
    last_seen_at        TEXT NOT NULL
);
CREATE INDEX idx_models_key ON models (key);
CREATE INDEX idx_models_path ON models (path);

-- Instance records (docs/architecture.md 搂3 "Instance").
-- `state` is the actual lifecycle state; `desired_state` is the state an
-- in-flight operation drives the instance toward (docs/architecture.md 搂9:
-- "operation銆乮nstance desired state 鍜?audit event 鍚屾鎻愪氦").
-- `pid` / `port` are only trusted once the process supervisor confirms them.
CREATE TABLE instances (
    instance_id     TEXT PRIMARY KEY,
    model_id        TEXT NOT NULL REFERENCES models (id) ON DELETE CASCADE,
    runtime_id      TEXT NOT NULL REFERENCES runtimes (id),
    load_config     TEXT NOT NULL,
    state           TEXT NOT NULL
        CONSTRAINT instance_state_valid
        CHECK (state IN ('unloaded', 'queued', 'loading', 'ready',
                         'draining', 'unloading', 'failed', 'crashed')),
    desired_state   TEXT NOT NULL
        CONSTRAINT instance_desired_state_valid
        CHECK (desired_state IN ('unloaded', 'queued', 'loading', 'ready',
                                 'draining', 'unloading', 'failed', 'crashed')),
    pid             INTEGER,
    port            INTEGER CHECK (port IS NULL OR (port BETWEEN 1 AND 65535)),
    device_ids      TEXT NOT NULL DEFAULT '[]',
    started_at      TEXT,
    last_used_at    TEXT,
    active_requests INTEGER NOT NULL DEFAULT 0 CHECK (active_requests >= 0),
    health          TEXT,
    failure         TEXT,
    created_at      TEXT NOT NULL,
    updated_at      TEXT NOT NULL
);
CREATE INDEX idx_instances_model ON instances (model_id);
CREATE INDEX idx_instances_runtime ON instances (runtime_id);
-- Partial index serving the docs/architecture.md 搂8 restart-recovery query.
CREATE INDEX idx_instances_state_nonterminal
    ON instances (state)
    WHERE state IN ('queued', 'loading', 'ready', 'draining', 'unloading');

-- Load / unload / rescan operations (docs/architecture.md 搂3 "Operation").
-- `error` is the structured `OperationError` JSON; `result` the structured
-- result payload JSON (docs/api.md 搂5).
CREATE TABLE operations (
    operation_id TEXT PRIMARY KEY,
    kind         TEXT NOT NULL CHECK (kind IN ('load', 'unload', 'rescan')),
    state        TEXT NOT NULL
        CONSTRAINT operation_state_valid
        CHECK (state IN ('queued', 'running', 'succeeded', 'failed', 'cancelled')),
    instance_id  TEXT REFERENCES instances (instance_id) ON DELETE SET NULL,
    model_id     TEXT REFERENCES models (id) ON DELETE SET NULL,
    created_at   TEXT NOT NULL,
    finished_at  TEXT,
    error        TEXT,
    result       TEXT
);
CREATE INDEX idx_operations_state ON operations (state);
CREATE INDEX idx_operations_instance ON operations (instance_id);
CREATE INDEX idx_operations_model ON operations (model_id);

-- Append-only audit / event log (docs/architecture.md 搂9, docs/api.md 搂5 SSE

    kind         TEXT NOT NULL,
    subject_type TEXT,
    subject_id   TEXT,
    operation_id TEXT,
    payload      TEXT NOT NULL DEFAULT '{}',
    created_at   TEXT NOT NULL
);
CREATE INDEX idx_audit_events_created ON audit_events (created_at);
CREATE INDEX idx_audit_events_operation ON audit_events (operation_id);
